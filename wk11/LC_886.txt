class Solution {
public:
    bool possibleBipartition(int n, vector<vector<int>>& dislikes) {
        vector<vector<int>> graph(n + 1);
        for (int i = 0; i < dislikes.size(); i++) {
            int u = dislikes[i][0];
            int v = dislikes[i][1];
            graph[u].push_back(v);
            graph[v].push_back(u);
        }

        vector<int> color(n + 1, 0);

        for (int i = 1; i <= n; i++) {
            if (color[i] != 0) {
                continue;
            }

            queue<int> q;
            q.push(i);
            color[i] = 1;

            while (q.empty() == false) {
                int current = q.front();
                q.pop();

                for (int j = 0; j < graph[current].size(); j++) {
                    int neighbor = graph[current][j];

                    if (color[neighbor] == color[current]) {
                        return false;
                    }

                    if (color[neighbor] == 0) {
                        if (color[current] == 1) {
                            color[neighbor] = 2;
                        } else {
                            color[neighbor] = 1;
                        }
                        q.push(neighbor);
                    }
                }
            }
        }

        return true;
    }
};
